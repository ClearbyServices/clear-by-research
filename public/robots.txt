import { MetadataRoute } from "next";

export default function robots(): MetadataRoute.Robots {
  return {
    rules: [
      {
        // Standard rules for all search engines (Google, Bing, etc.)
        userAgent: "*",
        allow: "/",
        disallow: [
          "/leaddesk/", // Keeps your private CRM totally hidden
          "/search/",   // Blocks search result pages
          "/*?",        // Blocks URLs with query params (like ?utm_ or ?s=) to prevent duplicate content
        ],
      },
      {
        // Block AI scraper bots from training on your content
        userAgent: ["GPTBot", "CCBot"],
        disallow: ["/"],
      },
    ],
    sitemap: "https://www.clearbyresearch.com/sitemap.xml",
  };
}
